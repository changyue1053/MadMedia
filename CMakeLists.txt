cmake_minimum_required(VERSION 3.16)
project(MadMedia VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
option(MADMEDIA_BUILD_TESTS "Build native tests" ON)
option(MADMEDIA_BUILD_TOOLS "Build developer tools" OFF)
option(MADMEDIA_WITH_MADOBS "Build optional header-only MadObs adapter" OFF)
option(IMMD_ASAN "Enable Address+UB sanitizers" OFF)
option(IMMD_TSAN "Enable Thread sanitizer" OFF)
option(IMMD_FUZZ "Build libFuzzer target" OFF)
if(IMMD_ASAN AND IMMD_TSAN)
 message(FATAL_ERROR "ASan and TSan are mutually exclusive")
endif()
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
find_package(SQLite3 REQUIRED)
file(GLOB_RECURSE SOURCES CONFIGURE_DEPENDS core/*.cpp)
add_library(madmedia_core STATIC ${SOURCES})
add_library(MadMedia::core ALIAS madmedia_core)
add_library(immd_core ALIAS madmedia_core)
set_target_properties(madmedia_core PROPERTIES EXPORT_NAME core POSITION_INDEPENDENT_CODE ON)
target_compile_features(madmedia_core PUBLIC cxx_std_17)
target_compile_options(madmedia_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_include_directories(madmedia_core PUBLIC
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
 $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
 $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/madmedia/internal>)
target_link_libraries(madmedia_core PUBLIC CURL::libcurl SQLite::SQLite3 Threads::Threads)
if(IMMD_ASAN)
 target_compile_options(madmedia_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(madmedia_core PUBLIC -fsanitize=address,undefined)
endif()
if(IMMD_TSAN)
 target_compile_options(madmedia_core PUBLIC -fsanitize=thread -fno-omit-frame-pointer)
 target_link_options(madmedia_core PUBLIC -fsanitize=thread)
endif()
# Preserve upstream warning isolation, but DO NOT disable ASan/UBSan coverage.
set_source_files_properties(core/pipeline/image/stb_impl.cpp PROPERTIES COMPILE_OPTIONS "-w")
if(MADMEDIA_WITH_MADOBS)
 find_path(MADOBS_INCLUDE_DIR madobs/observer.h REQUIRED)
 add_library(madmedia_obs INTERFACE)
 add_library(MadMedia::obs ALIAS madmedia_obs)
 target_link_libraries(madmedia_obs INTERFACE madmedia_core)
 target_include_directories(madmedia_obs INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/adapters/obs/include ${MADOBS_INCLUDE_DIR})
endif()
if(MADMEDIA_BUILD_TESTS)
 enable_testing()
 file(GLOB UNIT_SRC CONFIGURE_DEPENDS tests/unit_*.cpp)
 add_executable(madmedia_tests tests/test_main.cpp ${UNIT_SRC})
 target_link_libraries(madmedia_tests PRIVATE madmedia_core)
 add_test(NAME unit COMMAND madmedia_tests)
 add_executable(madmedia_e2e tests/e2e_main.cpp)
 target_link_libraries(madmedia_e2e PRIVATE madmedia_core)
 add_test(NAME e2e COMMAND madmedia_e2e)
 set_tests_properties(e2e PROPERTIES ENVIRONMENT "IMMD_MOCKCTL=${CMAKE_CURRENT_SOURCE_DIR}/testkit/mockctl.py" TIMEOUT 300)
 add_executable(madmedia_observer_test tests/sdk/observer_test.cpp)
 target_link_libraries(madmedia_observer_test PRIVATE madmedia_core)
 add_test(NAME sdk_observer COMMAND madmedia_observer_test)
 if(MADMEDIA_WITH_MADOBS)
  add_executable(madmedia_obs_adapter_test tests/sdk/obs_adapter_test.cpp)
  target_link_libraries(madmedia_obs_adapter_test PRIVATE madmedia_obs)
  add_test(NAME sdk_obs_adapter COMMAND madmedia_obs_adapter_test)
 endif()
endif()
if(MADMEDIA_BUILD_TOOLS)
 add_executable(madmedia-cli tools/immd-cli/main.cpp)
 target_link_libraries(madmedia-cli PRIVATE madmedia_core)
 add_executable(madmedia-bench tools/immd-bench/main.cpp)
 target_link_libraries(madmedia-bench PRIVATE madmedia_core)
endif()
if(IMMD_FUZZ)
 add_executable(madmedia_fuzz tests/fuzz/fuzz_all.cpp)
 target_link_libraries(madmedia_fuzz PRIVATE madmedia_core)
 target_compile_options(madmedia_fuzz PRIVATE -fsanitize=fuzzer,address)
 target_link_options(madmedia_fuzz PRIVATE -fsanitize=fuzzer,address)
endif()
install(TARGETS madmedia_core EXPORT MadMediaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY core/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/madmedia/internal/core FILES_MATCHING PATTERN "*.h")
install(DIRECTORY adapters/obs/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT MadMediaTargets NAMESPACE MadMedia:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MadMedia)
configure_package_config_file(cmake/MadMediaConfig.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/MadMediaConfig.cmake INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MadMedia)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/MadMediaConfigVersion.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MadMediaConfig.cmake ${CMAKE_CURRENT_BINARY_DIR}/MadMediaConfigVersion.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MadMedia)
install(DIRECTORY LICENSES/ DESTINATION ${CMAKE_INSTALL_DATADIR}/MadMedia/LICENSES)
install(FILES THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/MadMedia)
install(FILES third_party/stb/stb_image.h third_party/stb/stb_image_write.h third_party/stb/stb_image_resize2.h DESTINATION ${CMAKE_INSTALL_DATADIR}/MadMedia/LICENSES/stb)
if(MADMEDIA_BUILD_TESTS AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
 add_executable(madmedia_thread_failure_test tests/sdk/thread_failure_test.cpp)
 target_link_libraries(madmedia_thread_failure_test PRIVATE madmedia_core)
 target_link_libraries(madmedia_thread_failure_test PRIVATE ${CMAKE_DL_LIBS})
 foreach(n RANGE 1 7)
  add_test(NAME sdk_thread_failure_${n} COMMAND madmedia_thread_failure_test ${n})
  set_tests_properties(sdk_thread_failure_${n} PROPERTIES TIMEOUT 10)
 endforeach()
 add_executable(madmedia_callback_query_test tests/sdk/callback_query_test.cpp)
 target_link_libraries(madmedia_callback_query_test PRIVATE madmedia_core)
 add_test(NAME sdk_callback_query COMMAND madmedia_callback_query_test)
 set_tests_properties(sdk_callback_query PROPERTIES TIMEOUT 10)
endif()

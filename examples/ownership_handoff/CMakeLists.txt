cmake_minimum_required(VERSION 3.16)
project(MadMediaInstalledOwnership LANGUAGES C CXX)
if(ROLE STREQUAL "owner")
  set(sdk_target MadMedia::ffi)
  add_executable(ownership owner.c)
  set_target_properties(ownership PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF LINKER_LANGUAGE CXX)
elseif(ROLE STREQUAL "contender")
  set(sdk_target MadMedia::core)
  add_executable(ownership contender.cpp)
else()
  message(FATAL_ERROR "unrecognized role")
endif()
find_package(MadMedia CONFIG REQUIRED PATHS "${INSTALLED}/lib/cmake/MadMedia" NO_DEFAULT_PATH)
if(TARGET MadMedia::obs)
  message(FATAL_ERROR "observer must not be present")
endif()
# Both roles must use the real relocated release archives, even when the
# package or dependencies could also be installed elsewhere on the runner.
foreach(lib IN ITEMS core ffi)
  get_target_property(archive MadMedia::${lib} IMPORTED_LOCATION_RELEASE)
  if(NOT archive STREQUAL "${INSTALLED}/lib/libmadmedia_${lib}.a" OR NOT EXISTS "${archive}" OR IS_SYMLINK "${archive}")
    message(FATAL_ERROR "not the relocated release archive")
  endif()
endforeach()
target_link_libraries(ownership PRIVATE ${sdk_target})
target_compile_options(ownership PRIVATE -Wall -Wextra -Wpedantic -Werror)

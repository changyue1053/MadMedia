cmake_minimum_required(VERSION 3.16)
project(MadMediaMultiApp LANGUAGES C CXX)
option(WITH_OBS "Consume optional MadObs observer contract" OFF)
if(WITH_OBS)
 find_package(MadMedia CONFIG REQUIRED COMPONENTS Obs)
else()
 find_package(MadMedia CONFIG REQUIRED)
endif()
add_executable(app_a app_a.cpp)
target_compile_features(app_a PRIVATE cxx_std_17)
target_link_libraries(app_a PRIVATE MadMedia::core)
if(WITH_OBS)
 target_compile_definitions(app_a PRIVATE WITH_OBS=1)
 target_link_libraries(app_a PRIVATE MadMedia::obs)
endif()
add_executable(app_b app_b.c)
target_link_libraries(app_b PRIVATE MadMedia::ffi)
